## Research Notes

### WASAPI COM Interop (from 05-RESEARCH.md §2.1)

Required COM interfaces:
- `IMMDeviceEnumerator` — enumerates audio endpoints
- `IMMDevice` — represents an audio device
- `IAudioEndpointVolume` — controls volume/mute

COM threading: **STA apartment state** required. Use `[STAThread]` or `Thread.SetApartmentState(ApartmentState.STA)`.

```csharp
// COM interface definitions needed
[ComImport, Guid("A95664D2-9614-4F35-A746-DE8DB63617E6"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
public interface IMMDeviceEnumerator
{
    int EnumAudioEndpoints(int dataFlow, int stateMask, out IMMDeviceCollection devices);
    int GetDefaultAudioEndpoint(int dataFlow, int role, out IMMDevice device);
}

[ComImport, Guid("D666063F-1587-4E43-81F1-B948E807363F"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
public interface IMMDevice
{
    int Activate(ref Guid iid, int clsCtx, IntPtr activationParams, [MarshalAs(UnmanagedType.IUnknown)] out object interfacePointer);
    int OpenPropertyStore(int accessMask, out IntPtr propertyStore);
    int GetId([MarshalAs(UnmanagedType.LPWStr)] out string id);
    int GetState(out int state);
}

[ComImport, Guid("5CDF2C82-841E-4546-9722-0CF74078229A"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
public interface IAudioEndpointVolume
{
    int RegisterControlChangeNotify(IntPtr notify);
    int UnregisterControlChangeNotify(IntPtr notify);
    int GetChannelCount(out int count);
    int SetMasterVolumeLevel(float level, ref Guid eventContext);
    int SetMasterVolumeLevelScalar(float level, ref Guid eventContext);
    int GetMasterVolumeLevel(out float level);
    int GetMasterVolumeLevelScalar(out float level);
    int SetChannelVolumeLevel(int channel, float level, ref Guid eventContext);
    int SetChannelVolumeLevelScalar(int channel, float level, ref Guid eventContext);
    int GetChannelVolumeLevel(int channel, out float level);
    int GetChannelVolumeLevelScalar(int channel, out float level);
    int SetMute(bool mute, ref Guid eventContext);
    int GetMute(out bool mute);
}
```

### Registry Paths (from 05-RESEARCH.md §2.2)

| Setting | Hive | Key Path | Value Name | Enabled | Disabled |
|---------|------|----------|------------|---------|----------|
| Spatial Audio | HKLM | `SOFTWARE\Microsoft\Windows\CurrentVersion\Audio` | `SpatialAudioEnabled` | 1 | 0 |
| Bass Boost | HKLM | `SOFTWARE\Microsoft\Windows\CurrentVersion\Audio` | `BassBoostEnabled` | 1 | 0 |
| Loudness Equalization | HKLM | `SOFTWARE\Microsoft\Windows\CurrentVersion\Audio` | `LoudnessEqualizationEnabled` | 1 | 0 |

### Key Observations

- **WASAPI COM interop** requires STA threading — must use a dedicated STA thread or `[STAThread]`
- **Audio device enumeration** uses `IMMDeviceEnumerator.EnumAudioEndpoints()`
- **Volume control** uses `IAudioEndpointVolume` on the default endpoint
- **Spatial audio** is a simple registry DWORD under `SOFTWARE\Microsoft\Windows\CurrentVersion\Audio`
- **Audio enhancements** (bass boost, loudness equalization) are also registry DWORDs
- **SoundService** wraps both WASAPI COM interop and registry access
- **SoundPage** uses a custom layout (not TweakList) because it has device list + volume slider + toggles

## Design

### Architecture

1. **SoundService** — wraps WASAPI COM interop for device enumeration and volume, plus registry for spatial audio
2. **TweakCatalog.Sound** — static catalog of audio enhancement toggle definitions
3. **SoundPageViewModel** — orchestrates device list, volume, and enhancement toggles
4. **SoundPage** — custom layout with device list, volume slider, and enhancement toggles

```csharp
using System.Runtime.InteropServices;
using Microsoft.Extensions.Logging;
using Microsoft.Win32;

namespace VainTools.App.Services;

public sealed class SoundService : ISoundService
{
    private readonly IRegistryTweakService _registry;
    private readonly ILogger<SoundService> _logger;

    // COM interface GUIDs
    private static readonly Guid IMMDeviceEnumeratorGuid = new("A95664D2-9614-4F35-A746-DE8DB63617E6");
    private static readonly Guid IMMDeviceGuid = new("D666063F-1587-4E43-81F1-B948E807363F");
    private static readonly Guid IAudioEndpointVolumeGuid = new("5CDF2C82-841E-4546-9722-0CF74078229A");

    public SoundService(IRegistryTweakService registry, ILogger<SoundService> logger)
    {
        _registry = registry;
        _logger = logger;
    }

    public Task<IReadOnlyList<AudioDevice>> GetOutputDevicesAsync()
    {
        // WASAPI requires STA threading
        var tcs = new TaskCompletionSource<IReadOnlyList<AudioDevice>>();
        var thread = new Thread(() =>
        {
            try
            {
                var enumerator = (IMMDeviceEnumerator)new MMDeviceEnumerator();
                enumerator.EnumAudioEndpoints(0, 1, out var collection); // 0 = eRender, 1 = DEVICE_STATE_ACTIVE
                collection.GetCount(out int count);
                var devices = new List<AudioDevice>();
                for (int i = 0; i < count; i++)
                {
                    collection.Item(i, out var device);
                    device.GetId(out string id);
                    // Get friendly name from property store
                    device.OpenPropertyStore(0, out var propStore);
                    // ... read PKEY_Device_FriendlyName
                    var name = GetFriendlyName(propStore) ?? "Unknown Device";
                    // Check if default
                    enumerator.GetDefaultAudioEndpoint(0, 0, out var defaultDevice);
                    defaultDevice.GetId(out string defaultId);
                    var isDefault = id == defaultId;
                    Marshal.ReleaseComObject(defaultDevice);
                    Marshal.ReleaseComObject(device);
                    devices.Add(new AudioDevice(id, name, isDefault));
                }
                Marshal.ReleaseComObject(collection);
                Marshal.ReleaseComObject(enumerator);
                tcs.SetResult(devices);
            }
            catch (Exception ex)
            {
                tcs.SetException(ex);
            }
        });
        thread.SetApartmentState(ApartmentState.STA);
        thread.Start();
        return tcs.Task;
    }

    public Task<float> GetVolumeAsync()
    {
        var tcs = new TaskCompletionSource<float>();
        var thread = new Thread(() =>
        {
            try
            {
                var enumerator = (IMMDeviceEnumerator)new MMDeviceEnumerator();
                enumerator.GetDefaultAudioEndpoint(0, 0, out var device);
                device.Activate(ref IAudioEndpointVolumeGuid, 1, IntPtr.Zero, out var volumeObj);
                var volume = (IAudioEndpointVolume)volumeObj;
                volume.GetMasterVolumeLevelScalar(out float level);
                Marshal.ReleaseComObject(volume);
                Marshal.ReleaseComObject(device);
                Marshal.ReleaseComObject(enumerator);
                tcs.SetResult(level * 100f);
            }
            catch (Exception ex)
            {
                tcs.SetException(ex);
            }
        });
        thread.SetApartmentState(ApartmentState.STA);
        thread.Start();
        return tcs.Task;
    }

    public Task SetVolumeAsync(float volume)
    {
        var tcs = new TaskCompletionSource();
        var thread = new Thread(() =>
        {
            try
            {
                var enumerator = (IMMDeviceEnumerator)new MMDeviceEnumerator();
                enumerator.GetDefaultAudioEndpoint(0, 0, out var device);
                device.Activate(ref IAudioEndpointVolumeGuid, 1, IntPtr.Empty, out var volumeObj);
                var volume = (IAudioEndpointVolume)volumeObj;
                volume.SetMasterVolumeLevelScalar(volume / 100f, ref Guid.Empty);
                Marshal.ReleaseComObject(volume);
                Marshal.ReleaseComObject(device);
                Marshal.ReleaseComObject(enumerator);
                tcs.SetResult();
            }
            catch (Exception ex)
            {
                tcs.SetException(ex);
            }
        });
        thread.SetApartmentState(ApartmentState.STA);
        thread.Start();
        return tcs.Task;
    }

    public Task<bool> GetMuteAsync()
    {
        var tcs = new TaskCompletionSource<bool>();
        var thread = new Thread(() =>
        {
            try
            {
                var enumerator = (IMMDeviceEnumerator)new MMDeviceEnumerator();
                enumerator.GetDefaultAudioEndpoint(0, 0, out var device);
                device.Activate(ref IAudioEndpointVolumeGuid, 1, IntPtr.Empty, out var volumeObj);
                var volume = (IAudioEndpointVolume)volumeObj;
                volume.GetMute(out bool mute);
                Marshal.ReleaseComObject(volume);
                Marshal.ReleaseComObject(device);
                Marshal.ReleaseComObject(enumerator);
                tcs.SetResult(mute);
            }
            catch (Exception ex)
            {
                tcs.SetException(ex);
            }
        });
        thread.SetApartmentState(ApartmentState.STA);
        thread.Start();
        return tcs.Task;
    }

    public Task SetMuteAsync(bool mute)
    {
        var tcs = new TaskCompletionSource();
        var thread = new Thread(() =>
        {
            try
            {
                var enumerator = (IMMDeviceEnumerator)new MMDeviceEnumerator();
                enumerator.GetDefaultAudioEndpoint(0, 0, out var device);
                device.Activate(ref IAudioEndpointVolumeGuid, 1, IntPtr.Empty, out var volumeObj);
                var volume = (IAudioEndpointVolume)volumeObj;
                volume.SetMute(mute, ref Guid.Empty);
                Marshal.ReleaseComObject(volume);
                Marshal.ReleaseComObject(device);
                Marshal.ReleaseComObject(enumerator);
                tcs.SetResult();
            }
            catch (Exception ex)
            {
                tcs.SetException(ex);
            }
        });
        thread.SetApartmentState(ApartmentState.STA);
        thread.Start();
        return tcs.Task;
    }

    public Task<bool> IsSpatialAudioEnabledAsync()
    {
        var value = _registry.ReadDword(
            RegistryHive.LocalMachine,
            @"SOFTWARE\Microsoft\Windows\CurrentVersion\Audio",
            "SpatialAudioEnabled");
        return Task.FromResult(value == 1);
    }

    public Task SetSpatialAudioAsync(bool enabled)
    {
        _registry.WriteDword(
            RegistryHive.LocalMachine,
            @"SOFTWARE\Microsoft\Windows\CurrentVersion\Audio",
            "SpatialAudioEnabled",
            enabled ? 1 : 0);
        return Task.CompletedTask;
    }

    private static string? GetFriendlyName(IntPtr propertyStore)
    {
        // Read PKEY_Device_FriendlyName from IPropertyStore
        // PKEY_Device_FriendlyName = {a45c254e-df1c-4efd-8020-67d146a850e0} 14
        // ...
        return null; // placeholder
    }
}
```

```csharp
using System.Collections.ObjectModel;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using Microsoft.Extensions.Logging;
using VainTools.App.Services;
using VainTools.Framework.Services;
using VainTools.Framework.ViewModels;

namespace VainTools.App.ViewModels;

public partial class SoundPageViewModel : ViewModelBase
{
    private readonly ISoundService _soundService;
    private readonly IRegistryTweakService _registry;
    private readonly IDialogService _dialogs;
    private readonly IInfoBarService _infoBar;
    private readonly ILogger<SoundPageViewModel> _logger;

    [ObservableProperty]
    public partial string StatusMessage { get; set; } = "Ready";

    [ObservableProperty]
    public partial bool IsElevated { get; set; }

    [ObservableProperty]
    public partial float Volume { get; set; } = 50f;

    [ObservableProperty]
    public partial bool IsMuted { get; set; }

    [ObservableProperty]
    public partial bool IsSpatialAudioEnabled { get; set; }

    public ObservableCollection<AudioDevice> Devices { get; } = [];
    public ObservableCollection<TweakToggleViewModel> EnhancementTweaks { get; } = [];

    public SoundPageViewModel(
        ISoundService soundService,
        IRegistryTweakService registry,
        IDialogService dialogs,
        IInfoBarService infoBar,
        ILogger<SoundPageViewModel> logger)
    {
        _soundService = soundService;
        _registry = registry;
        _dialogs = dialogs;
        _infoBar = infoBar;
        _logger = logger;
        Title = "Sound";
        IsElevated = registry.IsElevated;

        // Load enhancement tweaks
        foreach (var tweak in TweakCatalog.Sound)
        {
            var row = new TweakToggleViewModel(tweak, registry, logger);
            row.Changed += OnRowChanged;
            EnhancementTweaks.Add(row);
        }
    }

    [RelayCommand]
    public async Task RefreshDevicesAsync()
    {
        try
        {
            var devices = await _soundService.GetOutputDevicesAsync();
            Devices.Clear();
            foreach (var device in devices)
            {
                Devices.Add(device);
            }
            StatusMessage = $"{devices.Count} audio devices found";
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to get audio devices");
            StatusMessage = $"Could not get devices: {ex.Message}";
            _infoBar.ShowError("Device enumeration failed", ex.Message);
        }
    }

    [RelayCommand]
    public async Task RefreshVolumeAsync()
    {
        try
        {
            Volume = await _soundService.GetVolumeAsync();
            IsMuted = await _soundService.GetMuteAsync();
            StatusMessage = "Volume settings loaded";
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to load volume settings");
            StatusMessage = $"Could not load volume: {ex.Message}";
        }
    }

    [RelayCommand]
    public async Task SetVolumeAsync()
    {
        try
        {
            await _soundService.SetVolumeAsync(Volume);
            StatusMessage = $"Volume set to {Volume:F0}%";
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to set volume");
            StatusMessage = $"Could not set volume: {ex.Message}";
            _infoBar.ShowError("Volume change failed", ex.Message);
        }
    }

    [RelayCommand]
    public async Task ToggleMuteAsync()
    {
        try
        {
            await _soundService.SetMuteAsync(IsMuted);
            StatusMessage = IsMuted ? "Audio muted" : "Audio unmuted";
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to toggle mute");
            StatusMessage = $"Could not toggle mute: {ex.Message}";
            _infoBar.ShowError("Mute toggle failed", ex.Message);
        }
    }

    [RelayCommand]
    public async Task RefreshSpatialAudioAsync()
    {
        try
        {
            IsSpatialAudioEnabled = await _soundService.IsSpatialAudioEnabledAsync();
            StatusMessage = "Spatial audio setting loaded";
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to load spatial audio setting");
            StatusMessage = $"Could not load spatial audio: {ex.Message}";
        }
    }

    [RelayCommand]
    public async Task SetSpatialAudioAsync()
    {
        try
        {
            await _soundService.SetSpatialAudioAsync(IsSpatialAudioEnabled);
            StatusMessage = IsSpatialAudioEnabled ? "Spatial audio enabled" : "Spatial audio disabled";
            _infoBar.ShowSuccess("Setting saved", $"Spatial audio has been {(IsSpatialAudioEnabled ? "enabled" : "disabled")}.");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to set spatial audio");
            StatusMessage = $"Could not set spatial audio: {ex.Message}";
            _infoBar.ShowError("Save failed", ex.Message);
        }
    }

    [RelayCommand]
    public void Refresh()
    {
        foreach (var row in EnhancementTweaks)
        {
            row.Refresh();
        }
        _ = RefreshDevicesAsync();
        _ = RefreshVolumeAsync();
        _ = RefreshSpatialAudioAsync();
    }

    private void OnRowChanged(object? sender, EventArgs e)
    {
        if (sender is TweakToggleViewModel row && !string.IsNullOrEmpty(row.LastError))
        {
            _infoBar.ShowError($"Could not change \"{row.Name}\"", row.LastError!);
            row.LastError = null;
        }
        UpdateStatus();
    }

    private void UpdateStatus()
    {
        var on = EnhancementTweaks.Count(t => t.IsOn);
        StatusMessage = $"{on} of {EnhancementTweaks.Count} audio enhancements enabled";
    }
}
```
