# Plan 05-03: Affinity Service, Startup Service & Pages

## Goal

Implement the Affinity page with CPU affinity management for processes (P/Invoke) and the
Startup page with Run-key entries (registry) and scheduled tasks (WMI).

**Decisions:** D-08 (P/Invoke for SetProcessAffinityMask / GetProcessAffinityMask), D-09
(AffinityService wraps P/Invoke), D-10 (registry for Run-key entries), D-11 (WMI for scheduled
tasks), D-12 (StartupService wraps registry + WMI), D-14 (custom layouts for Affinity and Startup)

### Key Observations

- **Affinity** requires P/Invoke — no pure .NET API for CPU affinity
- **Process enumeration** uses `EnumProcesses` + `GetModuleBaseName` from psapi.dll
- **Affinity masks** are `IntPtr` — must handle 64-bit masks correctly
- **System processes** (PID 0, 4) must be skipped — cannot open or modify affinity
- **Run-key entries** are simple registry string values — easy to read/write
- **Scheduled tasks** use WMI — the `Microsoft.Win32.TaskScheduler` NuGet package is recommended over raw WMI
- **AffinityService** wraps P/Invoke calls
- **StartupService** wraps registry + WMI
- **AffinityPage** and **StartupPage** both use custom layouts

```csharp
using System.ComponentModel;
using System.Runtime.InteropServices;
using System.Text;
using Microsoft.Extensions.Logging;

namespace VainTools.App.Services;

public sealed class AffinityService : IAffinityService
{
    private readonly ILogger<AffinityService> _logger;

    // P/Invoke constants
    private const uint PROCESS_QUERY_INFORMATION = 0x0400;
    private const uint PROCESS_SET_INFORMATION = 0x0200;

    public AffinityService(ILogger<AffinityService> logger)
    {
        _logger = logger;
    }

    public Task<IReadOnlyList<ProcessInfo>> GetProcessesAsync()
    {
        var processes = new List<ProcessInfo>();
        var processIds = new int[1024];
        int bytesReturned;

        if (!EnumProcesses(processIds, processIds.Length * sizeof(int), out bytesReturned))
        {
            throw new Win32Exception(Marshal.GetLastWin32Error(), "EnumProcesses failed");
        }

        int count = bytesReturned / sizeof(int);
        for (int i = 0; i < count; i++)
        {
            int pid = processIds[i];
            if (pid == 0 || pid == 4) continue; // Skip System Idle Process and System

            var handle = OpenProcess(PROCESS_QUERY_INFORMATION, false, pid);
            if (handle == IntPtr.Zero) continue;

            try
            {
                var sb = new StringBuilder(256);
                if (GetModuleBaseName(handle, IntPtr.Zero, sb, sb.Capacity) > 0)
                {
                    processes.Add(new ProcessInfo(pid, sb.ToString()));
                }
            }
            finally
            {
                CloseHandle(handle);
            }
        }

        return Task.FromResult<IReadOnlyList<ProcessInfo>>(processes);
    }

    public Task<ulong> GetProcessAffinityAsync(int processId)
    {
        var handle = OpenProcess(PROCESS_QUERY_INFORMATION, false, processId);
        if (handle == IntPtr.Zero)
            throw new Win32Exception(Marshal.GetLastWin32Error(), $"OpenProcess failed for PID {processId}");

        try
        {
            if (!GetProcessAffinityMask(handle, out IntPtr processMask, out _))
                throw new Win32Exception(Marshal.GetLastWin32Error(), "GetProcessAffinityMask failed");

            return Task.FromResult((ulong)processMask);
        }
        finally
        {
            CloseHandle(handle);
        }
    }

    public Task SetProcessAffinityAsync(int processId, ulong affinityMask)
    {
        var handle = OpenProcess(PROCESS_SET_INFORMATION, false, processId);
        if (handle == IntPtr.Zero)
            throw new Win32Exception(Marshal.GetLastWin32Error(), $"OpenProcess failed for PID {processId}");

        try
        {
            if (!SetProcessAffinityMask(handle, (IntPtr)affinityMask))
                throw new Win32Exception(Marshal.GetLastWin32Error(), "SetProcessAffinityMask failed");

            return Task.CompletedTask;
        }
        finally
        {
            CloseHandle(handle);
        }
    }

    public Task<ulong> GetSystemAffinityAsync()
    {
        var handle = OpenProcess(PROCESS_QUERY_INFORMATION, false, Environment.ProcessId);
        if (handle == IntPtr.Zero)
            throw new Win32Exception(Marshal.GetLastWin32Error(), "OpenProcess failed");

        try
        {
            if (!GetProcessAffinityMask(handle, out _, out IntPtr systemMask))
                throw new Win32Exception(Marshal.GetLastWin32Error(), "GetProcessAffinityMask failed");

            return Task.FromResult((ulong)systemMask);
        }
        finally
        {
            CloseHandle(handle);
        }
    }

    [DllImport("kernel32.dll")]
    private static extern IntPtr OpenProcess(uint processAccess, bool inheritHandle, int processId);

    [DllImport("kernel32.dll")]
    private static extern bool GetProcessAffinityMask(IntPtr processHandle, out IntPtr processAffinityMask, out IntPtr systemAffinityMask);

    [DllImport("kernel32.dll")]
    private static extern bool SetProcessAffinityMask(IntPtr processHandle, IntPtr processAffinityMask);

    [DllImport("kernel32.dll")]
    private static extern bool CloseHandle(IntPtr handle);

    [DllImport("psapi.dll")]
    private static extern int EnumProcesses(int[] processIds, int size, out int bytesReturned);

    [DllImport("psapi.dll", SetLastError = true)]
    private static extern int GetModuleBaseName(IntPtr processHandle, IntPtr module, StringBuilder baseName, int size);
}
```

```csharp
using Microsoft.Extensions.Logging;
using Microsoft.Win32;
using System.Management;

namespace VainTools.App.Services;

public sealed class StartupService : IStartupService
{
    private readonly ILogger<StartupService> _logger;

    public StartupService(ILogger<StartupService> logger)
    {
        _logger = logger;
    }

    public Task<IReadOnlyList<StartupEntry>> GetRunKeyEntriesAsync()
    {
        var entries = new List<StartupEntry>();

        // HKCU Run
        ReadRunKeyEntries(RegistryHive.CurrentUser, @"SOFTWARE\Microsoft\Windows\CurrentVersion\Run", "HKCU", entries);
        // HKLM Run
        ReadRunKeyEntries(RegistryHive.LocalMachine, @"SOFTWARE\Microsoft\Windows\CurrentVersion\Run", "HKLM", entries);
        // HKCU RunOnce
        ReadRunKeyEntries(RegistryHive.CurrentUser, @"SOFTWARE\Microsoft\Windows\CurrentVersion\RunOnce", "HKCU", entries);
        // HKLM RunOnce
        ReadRunKeyEntries(RegistryHive.LocalMachine, @"SOFTWARE\Microsoft\Windows\CurrentVersion\RunOnce", "HKLM", entries);
        // WOW6432Node Run
        ReadRunKeyEntries(RegistryHive.LocalMachine, @"SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Run", "HKLM", entries);

        return Task.FromResult<IReadOnlyList<StartupEntry>>(entries);
    }

    private static void ReadRunKeyEntries(RegistryHive hive, string keyPath, string hiveName, List<StartupEntry> entries)
    {
        using var key = RegistryKey.OpenBaseKey(hive, RegistryView.Registry64).OpenSubKey(keyPath);
        if (key == null) return;

        foreach (var valueName in key.GetValueNames())
        {
            var command = key.GetValue(valueName)?.ToString() ?? string.Empty;
            entries.Add(new StartupEntry(valueName, command, hiveName, keyPath));
        }
    }

    public Task AddRunKeyEntryAsync(string name, string command, bool isLocalMachine)
    {
        var hive = isLocalMachine ? RegistryHive.LocalMachine : RegistryHive.CurrentUser;
        var keyPath = @"SOFTWARE\Microsoft\Windows\CurrentVersion\Run";

        using var key = RegistryKey.OpenBaseKey(hive, RegistryView.Registry64).OpenSubKey(keyPath, true)
            ?? RegistryKey.OpenBaseKey(hive, RegistryView.Registry64).CreateSubKey(keyPath);

        key.SetValue(name, command, RegistryValueKind.String);
        return Task.CompletedTask;
    }

    public Task RemoveRunKeyEntryAsync(string name, bool isLocalMachine)
    {
        var hive = isLocalMachine ? RegistryHive.LocalMachine : RegistryHive.CurrentUser;
        var keyPath = @"SOFTWARE\Microsoft\Windows\CurrentVersion\Run";

        using var key = RegistryKey.OpenBaseKey(hive, RegistryView.Registry64).OpenSubKey(keyPath, true);
        if (key != null)
        {
            key.DeleteValue(name, false);
        }

        return Task.CompletedTask;
    }

    public Task<IReadOnlyList<ScheduledTask>> GetScheduledTasksAsync()
    {
        var tasks = new List<ScheduledTask>();

        try
        {
            // Use Microsoft.Win32.TaskScheduler NuGet package
            using var taskService = new Microsoft.Win32.TaskScheduler.TaskService();
            foreach (var task in taskService.RootFolder.EnumerateTasks())
            {
                bool hasStartupTrigger = false;
                try
                {
                    hasStartupTrigger = task.Definition.Triggers.Any(t =>
                        t.TriggerType == Microsoft.Win32.TaskScheduler.TaskTriggerType.Boot ||
                        t.TriggerType == Microsoft.Win32.TaskScheduler.TaskTriggerType.Logon);
                }
                catch { /* Some tasks may have inaccessible triggers */ }

                if (hasStartupTrigger)
                {
                    tasks.Add(new ScheduledTask(
                        task.Name,
                        task.Path,
                        task.State.ToString(),
                        true));
                }
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to enumerate scheduled tasks");
        }

        return Task.FromResult<IReadOnlyList<ScheduledTask>>(tasks);
    }

    public Task EnableScheduledTaskAsync(string taskName)
    {
        using var taskService = new Microsoft.Win32.TaskScheduler.TaskService();
        var task = taskService.RootFolder.GetTask(taskName);
        if (task != null)
        {
            task.Enabled = true;
        }
        return Task.CompletedTask;
    }

    public Task DisableScheduledTaskAsync(string taskName)
    {
        using var taskService = new Microsoft.Win32.TaskScheduler.TaskService();
        var task = taskService.RootFolder.GetTask(taskName);
        if (task != null)
        {
            task.Enabled = false;
        }
        return Task.CompletedTask;
    }
}
```

```csharp
using System.Collections.ObjectModel;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using Microsoft.Extensions.Logging;
using VainTools.App.Services;
using VainTools.Framework.Services;
using VainTools.Framework.ViewModels;

namespace VainTools.App.ViewModels;

public partial class AffinityPageViewModel : ViewModelBase
{
    private readonly IAffinityService _affinityService;
    private readonly IDialogService _dialogs;
    private readonly IInfoBarService _infoBar;
    private readonly ILogger<AffinityPageViewModel> _logger;

    [ObservableProperty]
    public partial string StatusMessage { get; set; } = "Ready";

    [ObservableProperty]
    public partial bool IsElevated { get; set; }

    [ObservableProperty]
    public partial ulong AffinityMask { get; set; }

    [ObservableProperty]
    public partial ulong SystemAffinityMask { get; set; }

    [ObservableProperty]
    public partial int CpuCount { get; set; }

    public ObservableCollection<ProcessInfo> Processes { get; } = [];

    public AffinityPageViewModel(
        IAffinityService affinityService,
        IDialogService dialogs,
        IInfoBarService infoBar,
        ILogger<AffinityPageViewModel> logger)
    {
        _affinityService = affinityService;
        _dialogs = dialogs;
        _infoBar = infoBar;
        _logger = logger;
        Title = "Affinity";
        IsElevated = Environment.Is64BitProcess; // Simplified elevation check
    }

    [RelayCommand]
    public async Task RefreshProcessesAsync()
    {
        try
        {
            var processes = await _affinityService.GetProcessesAsync();
            Processes.Clear();
            foreach (var process in processes.OrderBy(p => p.Name))
            {
                Processes.Add(process);
            }
            StatusMessage = $"{processes.Count} processes found";
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to get processes");
            StatusMessage = $"Could not get processes: {ex.Message}";
            _infoBar.ShowError("Process enumeration failed", ex.Message);
        }
    }

    [RelayCommand]
    public async Task RefreshAffinityAsync()
    {
        try
        {
            SystemAffinityMask = await _affinityService.GetSystemAffinityAsync();
            CpuCount = Environment.ProcessorCount;

            if (SelectedProcess != null)
            {
                AffinityMask = await _affinityService.GetProcessAffinityAsync(SelectedProcess.ProcessId);
            }

            StatusMessage = "Affinity settings loaded";
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to load affinity settings");
            StatusMessage = $"Could not load affinity: {ex.Message}";
        }
    }

    [RelayCommand]
    public async Task SetAffinityAsync()
    {
        try
        {
            if (SelectedProcess == null)
            {
                _infoBar.ShowWarning("No process selected", "Please select a process first.");
                return;
            }

            await _affinityService.SetProcessAffinityAsync(SelectedProcess.ProcessId, AffinityMask);
            StatusMessage = $"Affinity set for {SelectedProcess.Name}";
            _infoBar.ShowSuccess("Affinity updated", $"CPU affinity has been updated for {SelectedProcess.Name}.");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to set affinity");
            StatusMessage = $"Could not set affinity: {ex.Message}";
            _infoBar.ShowError("Affinity change failed", ex.Message);
        }
    }

    [RelayCommand]
    public void Refresh()
    {
        _ = RefreshProcessesAsync();
        _ = RefreshAffinityAsync();
    }

    partial void OnSelectedProcessChanged(ProcessInfo? value)
    {
        if (value != null)
        {
            _ = RefreshAffinityAsync();
```
